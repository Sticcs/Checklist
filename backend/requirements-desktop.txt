-r requirements.txt
pywebview
# pywebview's macOS backend is Cocoa/WebKit via PyObjC - pip doesn't pull
# these in automatically the way `pip install pywebview[cocoa]` would, and
# that extras syntax doesn't get skipped cleanly on other platforms, so
# they're listed directly here instead, gated by environment marker (a
# no-op install on Windows/Linux).
pyobjc-framework-Cocoa; sys_platform == "darwin"
pyobjc-framework-WebKit; sys_platform == "darwin"
pyobjc-framework-Quartz; sys_platform == "darwin"
pyinstaller
# Native OS notifications (Windows toast / macOS Notification Center) - see
# desktop.py's notify(). The Web Notification API the website uses isn't
# reliably available inside an embedded webview.
plyer
# plyer's macOS notification backend bridges Objective-C via pyobjus (Kivy's
# own bridge - a different package from the pyobjc-framework-* ones above,
# which pywebview's Cocoa backend uses instead), so it isn't covered by those.
pyobjus; sys_platform == "darwin"
